void handleModeRadio(ButtonEvent evt) {
  static uint32_t lastUpdate = 0;
  if (millis() - lastUpdate > 500) {
    radioFreq = radio.getFrequency();
    RADIO_INFO info;
    radio.checkRDS(); updateRDS(); radio.getRadioInfo(&info);
    radioStereo = info.stereo;
    lastUpdate = millis();
    drawRadioUI();
  }

  if (!evt.valid) return;

  if (evt.pin == BTN_B) {
    if (evt.isLong) {
      radio.setMute(true); saveSettings();
      appMode = MODE_FEATURES;
      drawFeaturesMenu(menuFeatSel);
      resetAllButtons();
      return;
    } else {
      radioVol = (radioVol + 1) % 16;
      radio.setVolume(radioVol); saveSettings();
    }
  } else if (evt.pin == BTN_C) {
    if (evt.isLong) {
      radio.seekDown(true);
    } else {
      radioFreq -= 10;
      if (radioFreq < 5000) radioFreq = 11500;
      radio.setFrequency(radioFreq); saveSettings();
    }
  } else if (evt.pin == BTN_D) {
    if (evt.isLong) {
      radio.seekUp(true);
    } else {
      radioFreq += 10;
      if (radioFreq > 11500) radioFreq = 5000;
      radio.setFrequency(radioFreq); saveSettings();
    }
  } else if (evt.pin == BTN_BOOT) {
    if (evt.isLong) {
      radio.seekUp(true);
    } else {
      radioMute = !radioMute;
      radio.setMute(radioMute); saveSettings();
    }
  }

  drawRadioUI();
}
